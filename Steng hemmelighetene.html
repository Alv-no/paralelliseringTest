Steng hemmelighetene

Når du er ferdig, kan ikke Claude Code lese nøklene dine, og den må spørre deg før den rører noe utenfor prosjektet du jobber i.

Gjelder Claude Code på Windows, med PowerShell eller Git Bash. I WSL fungerer deny-reglene, men ikke hooken.

  • STOPP~/.azure, ~/.aws, ~/.ssh, .env og secrets.* blir avvist, også i shell-kommandoer.
  • STOPPKommandoer som skriver ut lagrede tokens, som git credential fill, gh auth token og az account get-access-token. git push og gh pr create virker som før.
  • SPØRAlt utenfor prosjektmappa gir en dialog du godkjenner eller avslår.
  • OKArbeid inne i prosjektet går som før.

  1. Sjekk at Node er installert

    Hooken er et lite Node-skript. Kjør dette i PowerShell. Du trenger versjon 18 eller nyere.

    node --version

    Får du feil, installer Node LTS fra nodejs.org før du går videre.

  2. Lagre hooken

    Lag mappa og åpne den første fila i Notisblokk:

    New-Item -ItemType Directory -Force $HOME\.claude\hooks | Out-Null
notepad $HOME\.claude\hooks\path-guard.mjs

    Lim inn skriptet under, og lagre.

    Vis path-guard.mjs 
    // path-guard: PreToolUse-hook for Claude Code på Windows.
//
//   deny      -> blokkerer (hemmeligheter: ~/.azure, ~/.aws, ~/.ssh, .env, secrets.*)
//   utenfor   -> spør deg i en dialog
//   innenfor  -> går rett gjennom
//
// Innstillinger i path-guard.json ved siden av denne fila. Filverktøyene (Read, Write, Edit,
// Glob, Grep) sjekkes presist. Bash/PowerShell er fritekst: stier letes fram med mønstre, og en
// sti som bygges i en variabel slipper gjennom. Det er et sikkerhetsnett, ikke en vegg.
// Feiler skriptet selv, svarer det «ask» — aldri stille «ok».

import { readFileSync } from "node:fs";
import { dirname, join, win32 } from "node:path";
import { fileURLToPath } from "node:url";

const here = dirname(fileURLToPath(import.meta.url));

function expandEnv(s) {
  return s
    .replace(/%([A-Za-z_][A-Za-z0-9_]*)%/g, (m, n) => process.env[n] ?? m)
    .replace(/\$env:([A-Za-z_][A-Za-z0-9_]*)/gi, (m, n) => process.env[n] ?? process.env[n.toUpperCase()] ?? m)
    .replace(/\$\{([A-Za-z_][A-Za-z0-9_]*)\}/g, (m, n) => process.env[n] ?? m)
    .replace(/\$(HOME|USERPROFILE|TEMP|TMP|APPDATA|LOCALAPPDATA|PROGRAMDATA)\b/g, (m, n) =>
      n === "HOME" ? (process.env.USERPROFILE ?? m) : (process.env[n] ?? m),
    );
}

// Absolutt Windows-form: /c/x -> C:\x, ~ -> hjemmemappa, relativ -> mot cwd.
function normalize(p, cwd) {
  let s = expandEnv(p.trim().replace(/^["']|["']$/g, ""));
  if (/^~(?=[\\/]|$)/.test(s)) s = (process.env.USERPROFILE ?? "") + s.slice(1);
  const msys = s.match(/^\/([a-zA-Z])(?=\/|$)(.*)$/);
  if (msys) s = `${msys[1].toUpperCase()}:${msys[2] || "\\"}`;
  else if (/^\/(tmp|usr|etc|bin|mingw64|ucrt64|opt|var|home)(\/|$)/.test(s)) return `[Git Bash]${s}`;
  return win32.resolve(cwd || process.cwd(), s);
}

const toRoot = (r) => win32.resolve(expandEnv(r)).toLowerCase().replace(/[\\/]+$/, "");

function isInside(abs, roots) {
  const a = abs.toLowerCase().replace(/[\\/]+$/, "");
  return roots.some((r) => a === r || a.startsWith(r + "\\"));
}

// ".env", ".env.*", "secrets.*" -> regex mot filnavnet.
const nameMatcher = (globs) => {
  const res = globs.map(
    (g) => new RegExp("^" + g.replace(/[.+^${}()|[\]\\]/g, "\\$&").replace(/\*/g, ".*").replace(/\?/g, ".") + "$", "i"),
  );
  return (p) => res.some((re) => re.test(win32.basename(p.replace(/[\\/]+$/, ""))));
};

// Et tillatt skript der det står som kommandoen som kjøres (først, eller etter en tolk) fjernes.
// Står samme sti andre steder (`> skript.sh`), sjekkes den som vanlig.
function stripRunnable(cmd, runnable, cwd) {
  if (runnable.length === 0) return cmd;
  const start = String.raw`(^|(?:&&|\|\||[;&|\n(])\s*)`;
  const interp = String.raw`((?:(?:bash|sh|node|pwsh|powershell|python3?|&)(?:\.exe)?\s+(?:-(?:File|NoProfile|ExecutionPolicy\s+\S+)\s+)*)?)`;
  const token = String.raw`("[^"]+"|'[^']+'|[^\s;&|]+)`;
  return cmd.replace(new RegExp(start + interp + token, "gi"), (all, pre, int, tok) =>
    runnable.includes(normalize(tok, cwd).toLowerCase()) ? `${pre}${int}` : all,
  );
}

// Stikandidater i en shell-kommando. Bevisst grovt: heller et spørsmål for mye.
function pathsInCommand(cmd, isSecretName) {
  const found = new Set();
  // Venstresiden i en tilordning er et navn, ikke en sti: `$env:TEMP='C:\x'` skal sjekke C:\x.
  const text = expandEnv(
    cmd
      .replace(/\$env:[A-Za-z_][A-Za-z0-9_]*\s*=/gi, " ")
      .replace(/\b(?:export|set|setx)\s+[A-Za-z_][A-Za-z0-9_]*\s*=?/gi, " "),
  );
  const boundary = String.raw`(?:^|[\s"'=(<>|;&,])`;
  const patterns = [
    String.raw`[A-Za-z]:[\\/][^\s"'|;&<>()]*`, // C:\x
    String.raw`\\\\[^\s"'|;&<>()]+`, // \\server\share
    String.raw`/[a-zA-Z](?=/|[\s"']|$)[^\s"'|;&<>()]*`, // /c/x
    String.raw`/(?:tmp|usr|etc|bin|mingw64|ucrt64|opt|var|home)(?=/|[\s"']|$)[^\s"'|;&<>()]*`,
    String.raw`~(?=[\\/]|[\s"']|$)[^\s"'|;&<>()]*`, // ~ og ~/x
    String.raw`[^\s"'|;&<>()]*\.\.(?:[\\/][^\s"'|;&<>()]*)?`, // alt med ..
  ];
  for (const p of patterns) {
    for (const m of text.matchAll(new RegExp(boundary + "(" + p + ")", "g"))) {
      const s = m[1].replace(/[=,;:]+$/, ""); // `a.cs,` `x=` -> a.cs, x
      if (s) found.add(s);
    }
  }
  // Relative filnavn som `cat .env` har ingen skråstrek og fanges ikke over.
  for (const t of text.split(/[\s"'|;&<>(),=]+/)) if (t && isSecretName(t)) found.add(t);
  return [...found];
}

function pathsInFileTool(name, input) {
  const out = [];
  for (const key of ["file_path", "notebook_path", "path"]) {
    if (typeof input[key] === "string" && input[key]) out.push(input[key]);
  }
  if ((name === "Glob" || name === "Grep") && !input.path) {
    const pat = typeof input.pattern === "string" ? input.pattern : "";
    out.push(/^([A-Za-z]:[\\/]|\/[a-zA-Z]\/|~)/.test(pat) ? pat.replace(/[*?[{].*$/, "") || "." : ".");
  }
  return out;
}

function respond(decision, reason) {
  process.stdout.write(
    JSON.stringify({
      hookSpecificOutput: { hookEventName: "PreToolUse", permissionDecision: decision, permissionDecisionReason: reason },
    }),
  );
  process.exit(0);
}

try {
  const event = JSON.parse(readFileSync(0, "utf8"));
  const config = JSON.parse(readFileSync(join(here, "path-guard.json"), "utf8"));
  const cwd = event.cwd;
  const name = event.tool_name;
  const input = event.tool_input ?? {};

  const roots = (config.allowed ?? []).map(toRoot);
  // Prosjektet økta jobber i er alltid tillatt — men aldri en hel disk eller hele hjemmemappa.
  const projectDir = config.allowProjectDir !== false && (process.env.CLAUDE_PROJECT_DIR || cwd);
  if (projectDir) {
    const p = toRoot(projectDir);
    if (!/^[a-z]:$/.test(p) && p !== toRoot(process.env.USERPROFILE ?? "")) roots.push(p);
  }
  const readRoots = (config.readOnly ?? []).map(toRoot);
  const denyRoots = (config.deny ?? []).map(toRoot);
  const isSecretName = nameMatcher(config.denyNames ?? []);
  const runnable = (config.runnable ?? []).map((r) => normalize(r, cwd).toLowerCase());
  const alwaysOk = new Set((config.alwaysOk ?? []).map((s) => s.toLowerCase()));

  const isShell = name === "Bash" || name === "PowerShell";
  const readsOnly = name === "Read" || name === "Glob" || name === "Grep";

  // Tokens i Windows Legitimasjonsbehandling er ikke filer, så stisjekken ser dem aldri.
  // Kommandoene som skriver dem ut, stoppes på navn — uansett hvor i kommandoen de står.
  if (isShell) {
    const cmd = String(input.command ?? "");
    const hit = (config.denyCommands ?? []).find((re) => new RegExp(re, "i").test(cmd));
    if (hit) {
      respond(
        "deny",
        `⛔ BLOKKERT: HENTER UT TOKEN\n\npath-guard: kommandoen kan skrive ut et lagret token eller passord ` +
          `(treff på denyCommands: ${hit}).\nDette slippes ikke gjennom. Trenger du det, gjør det selv utenfor Claude.`,
      );
    }
  }
  const candidates = isShell
    ? pathsInCommand(stripRunnable(String(input.command ?? ""), runnable, cwd), isSecretName)
    : pathsInFileTool(name, input);

  const denied = [];
  const outside = [];
  for (const c of candidates) {
    if (alwaysOk.has(c.toLowerCase())) continue;
    const abs = normalize(c, cwd);
    if (abs.startsWith("[Git Bash]")) outside.push(abs);
    else if (isInside(abs, denyRoots) || isSecretName(abs)) denied.push(abs);
    else if (isInside(abs, roots)) continue;
    else if (isInside(abs, readRoots)) {
      if (!readsOnly) outside.push(`${abs} (kun lesetilgang)`);
    } else outside.push(abs);
  }
  if (isShell && cwd && !isInside(win32.resolve(cwd), roots)) outside.push(`${win32.resolve(cwd)} (arbeidsmappe)`);

  const list = (xs) => [...new Set(xs)].map((p) => `  - ${p}`).join("\n");
  if (denied.length > 0) {
    respond(
      "deny",
      `⛔ BLOKKERT: HEMMELIGHET\n\npath-guard: ${name} rører en sti på deny-lista:\n${list(denied)}\n` +
        `Dette slippes ikke gjennom. Trenger du det, gjør det selv utenfor Claude.`,
    );
  }
  if (outside.length === 0) process.exit(0);
  const headline = isShell
    ? "🔴 SKUMMEL SHELL-KOMMANDO (kan lese eller skrive)"
    : readsOnly
      ? "🔴 SKUMMEL LESING"
      : "🔴 SKUMMEL SKRIVING";
  respond(
    "ask",
    `${headline}\n\npath-guard: ${name} rører ${outside.length === 1 ? "en sti" : "stier"} utenfor tillatte mapper:\n` +
      `${list(outside)}\nRediger ${join(here, "path-guard.json")} for å endre hva som er tillatt.`,
  );
} catch (err) {
  respond("ask", `path-guard kunne ikke vurdere kallet (${err.message}) — spør for sikkerhets skyld.`);
}

  3. Koble den inn, og legg til deny-reglene

    Ta en kopi først, så kan du gå tilbake hvis noe blir feil:

    Copy-Item $HOME\.claude\settings.json $HOME\.claude\settings.json.bak -ErrorAction SilentlyContinue
notepad $HOME\.claude\settings.json

    Er fila tom eller ny, lim inn alt under. Har den innhold fra før, legg "permissions" og "hooks" inn ved siden av det som står der, med komma mellom. Står "permissions" der allerede, legg "deny"-lista inn i den. To "permissions" i samme fil gjør at den første forsvinner uten feilmelding.

    {
  "permissions": {
    "deny": [
      "Read(**/.env)",
      "Read(**/.env.*)",
      "Read(**/secrets.*)",
      "Read(~/.azure/**)",
      "Read(~/.aws/**)",
      "Read(~/.ssh/**)"
    ]
  },
  "hooks": {
    "PreToolUse": [
      {
        "matcher": "Read|Write|Edit|Glob|Grep|NotebookEdit|Bash|PowerShell",
        "hooks": [
          {
            "type": "command",
            "command": "node \"$HOME/.claude/hooks/path-guard.mjs\"",
            "timeout": 10,
            "onFailure": "block"
          }
        ]
      }
    ]
  }
}

    Deny-reglene stopper filverktøyene. Hooken stopper i tillegg shell-kommandoer som cat ~/.azure/config, som deny-reglene ikke ser.

    Innstillingene lastes når en økt starter. Åpne en ny Claude Code-økt i et prosjekt og be om disse fire tingene:

    Går noen av dem rett gjennom, sjekk at settings.json er gyldig JSON og at økta ble startet etter at du lagret.

Når dialogen dukker opp

Les stiene i dialogen. Skjønner du hvorfor Claude trenger dem, godkjenn. Er du i tvil, trykk Deny. I shell-kommandoer ser hooken bare teksten, så en kommando som bare nevner en sti, gir også dialog. Det er meningen.

Det samme gjelder blokkeringene: grep "gh auth token" README.md blir stoppet selv om den bare søker etter teksten. Trenger du en slik kommando, kjør den selv i terminalen.

Tilpass i path-guard.json

  • allowed: mapper Claude kan lese og skrive fritt i, i tillegg til prosjektet.
  • readOnly: mapper Claude kan lese, men ikke skrive i. ~/.claude står der, så Claude ikke kan slå av sin egen hook.
  • deny og denyNames: mapper og filnavn som alltid blokkeres.
  • denyCommands: mønstre for kommandoer som alltid blokkeres, som de som henter ut tokens fra Windows Legitimasjonsbehandling.
  • runnable: hele stier til skript som kan kjøres uten dialog, men ikke redigeres.

Endringer gjelder fra neste verktøykall. Hooken er et sikkerhetsnett, ikke en vegg: en sti som bygges i en variabel i et skript, slipper gjennom. Krasjer hooken, eller mangler Node, blokkeres alle verktøykall til det er fikset.
